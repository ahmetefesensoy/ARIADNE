cmake_minimum_required(VERSION 3.8)
project(redline_description)
find_package(ament_cmake REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_rover.py"
  --out "${GENERATED_DIR}" RESULT_VARIABLE generation_result)
if(NOT generation_result EQUAL 0)
  message(FATAL_ERROR "Rover description generation failed")
endif()
install(DIRECTORY "${GENERATED_DIR}/" DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/generate_rover.py DESTINATION lib/${PROJECT_NAME})
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_scout.py"
  --out "${GENERATED_DIR}/models/scout_1" RESULT_VARIABLE scout_generation_result)
if(NOT scout_generation_result EQUAL 0)
  message(FATAL_ERROR "SCENARIO scout generation failed")
endif()
install(PROGRAMS scripts/generate_scout.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(rover_contract test/test_description.py)
  ament_add_pytest_test(delivered_rover_contract test/test_delivered_rover.py)
endif()
ament_package()
