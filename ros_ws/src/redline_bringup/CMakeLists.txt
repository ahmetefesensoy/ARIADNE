cmake_minimum_required(VERSION 3.8)
project(redline_bringup)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(rviz_goal_observer src/rviz_goal_observer.cpp)
target_compile_features(rviz_goal_observer PUBLIC cxx_std_17)
ament_target_dependencies(rviz_goal_observer rclcpp geometry_msgs std_msgs)
target_link_libraries(rviz_goal_observer nlohmann_json::nlohmann_json)
install(TARGETS rviz_goal_observer DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/cmd_watchdog.py scripts/verify_navigation.py scripts/ground_filter.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(command_gate test/test_watchdog.py)
  ament_add_pytest_test(navigation_verifier test/test_navigation_verifier.py)
  ament_add_pytest_test(rviz_goal_observer test/test_goal_observer.py TIMEOUT 30)
  ament_add_pytest_test(collision_limit test/test_collision_limit.py TIMEOUT 30)
  ament_add_pytest_test(ground_filter test/test_ground_filter.py TIMEOUT 30)
endif()
ament_package()
